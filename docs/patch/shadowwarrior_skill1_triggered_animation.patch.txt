# ── 1) const 区新增 preload（角色级常量，建议放在 SHADOWWARRIOR_ATTACK_FOOT_GAPS 之后）──
#    注意：若后续还要加 skill2_triggered 等状态，建议改用与现有文件一致的
#    状态级命名（如 SHADOWWARRIOR_SKILL1_TRIGGERED_FOOT_GAPS），避免角色级常量重名。
const SHADOWWARRIOR_FOOT_GAPS = preload("res://data/foot_gaps/shadowwarrior_skill1_triggered_foot_gaps.gd")

# ── 2) get_config() 的 animations 字典中新增/替换一行 ──
"skill1_triggered": FrameAnimation.load_from_sprite_sheet(SHADOWWARRIOR_ANI_DIR + "skill1_triggered/sheet.png", 2, 2, 3, 0.1, false, _shadowwarrior_skill1_triggered_anchors()),

# ── 3) 新增锚点辅助函数（放在 _shadowwarrior_attack_anchors 之后）──
static func _shadowwarrior_skill1_triggered_anchors() -> Array:
	var anchors: Array = []
	for i in range(SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_SKILL1_TRIGGERED_FOOT.size()):
		anchors.append({
			"foot_gap": SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_SKILL1_TRIGGERED_FOOT[i],
			"head_gap": SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_SKILL1_TRIGGERED_HEAD[i],
			"center_dx": SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_SKILL1_TRIGGERED_CENTER[i],
			"content_w": SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_SKILL1_TRIGGERED_CONTENT_W[i],
			"content_h": SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_SKILL1_TRIGGERED_CONTENT_H[i],
		})
	return anchors
