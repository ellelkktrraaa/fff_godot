## c) 锚点辅助函数（追加到类内）
func _dragon_knight_idle_anchors() -> Array[Dictionary]:
    var anchors: Array[Dictionary] = []
    for i in DRAGON_KNIGHT_IDLE_FOOT.size():
        anchors.append({
            "foot_gap": DRAGON_KNIGHT_FOOT_GAPS.DRAGON_KNIGHT_IDLE_FOOT[i],
            "head_gap": DRAGON_KNIGHT_FOOT_GAPS.DRAGON_KNIGHT_IDLE_HEAD[i],
            "center_dx": DRAGON_KNIGHT_FOOT_GAPS.DRAGON_KNIGHT_IDLE_CENTER[i],
            "content_w": DRAGON_KNIGHT_FOOT_GAPS.DRAGON_KNIGHT_IDLE_CONTENT_W[i],
            "content_h": DRAGON_KNIGHT_FOOT_GAPS.DRAGON_KNIGHT_IDLE_CONTENT_H[i],
        })
    return anchors
