# --- 1) const 区新增（与 ROSE_ANI_DIR / ROSE_IDLE_FOOT_GAPS 等并列）---
const ROSE_FOOT_GAPS = preload("res://data/foot_gaps/rose_ult_foot_gaps.gd")

# --- 2) animations 字典中 "ult" 行替换为 ---
"ult": FrameAnimation.load_from_sprite_sheet(ROSE_ANI_DIR + "ult/sheet.png", 8, 7, 54, 0.1, false, _rose_ult_anchors()),

# --- 3) 新增锚点辅助函数（参照 _rose_idle_anchors 写法）---
static func _rose_ult_anchors() -> Array:
	var anchors: Array = []
	for i in range(ROSE_FOOT_GAPS.ROSE_ULT_FOOT.size()):
		anchors.append({
			"foot_gap": ROSE_FOOT_GAPS.ROSE_ULT_FOOT[i],
			"head_gap": ROSE_FOOT_GAPS.ROSE_ULT_HEAD[i],
			"center_dx": ROSE_FOOT_GAPS.ROSE_ULT_CENTER[i],
			"content_w": ROSE_FOOT_GAPS.ROSE_ULT_CONTENT_W[i],
			"content_h": ROSE_FOOT_GAPS.ROSE_ULT_CONTENT_H[i],
		})
	return anchors
