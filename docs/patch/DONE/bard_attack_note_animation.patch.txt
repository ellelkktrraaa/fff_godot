## 3) 新增锚点辅助函数（参考 _bard_attack_anchors()）：
static func _bard_attack_note_anchors() -> Array:
	var anchors := []
	for i in range(BARD_FOOT_GAPS.BARD_ATTACK_NOTE_FOOT.size()):
		anchors.append({
			"foot_gap": BARD_FOOT_GAPS.BARD_ATTACK_NOTE_FOOT[i],
			"head_gap": BARD_FOOT_GAPS.BARD_ATTACK_NOTE_HEAD[i],
			"center_dx": BARD_FOOT_GAPS.BARD_ATTACK_NOTE_CENTER[i],
			"content_w": BARD_FOOT_GAPS.BARD_ATTACK_NOTE_CONTENT_W[i],
			"content_h": BARD_FOOT_GAPS.BARD_ATTACK_NOTE_CONTENT_H[i],
		})
	return anchors
