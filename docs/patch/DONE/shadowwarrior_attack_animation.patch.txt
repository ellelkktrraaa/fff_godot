# ── shadowwarrior.gd 改动建议（attack 动画从 load_from_frames 迁移到 sprite-sheet）──

## a) const 区新增 preload（建议放在 SHADOWWARRIOR_ANI_DIR 常量附近）
const SHADOWWARRIOR_FOOT_GAPS = preload("res://data/foot_gaps/shadowwarrior_attack_foot_gaps.gd")

## b) animations 字典中 "attack" 一行替换为：
"attack": FrameAnimation.load_from_sprite_sheet(SHADOWWARRIOR_ANI_DIR + "attack/sheet.png", 4, 3, 10, 0.1, false, _shadowwarrior_attack_anchors()),

## c) 锚点辅助函数（可加在 get_config() 之后）
static func _shadowwarrior_attack_anchors() -> Array:
	var anchors: Array = []
	for i in range(SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_ATTACK_FOOT.size()):
		anchors.append({
			"foot_gap": SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_ATTACK_FOOT[i],
			"head_gap": SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_ATTACK_HEAD[i],
			"center_dx": SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_ATTACK_CENTER[i],
			"content_w": SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_ATTACK_CONTENT_W[i],
			"content_h": SHADOWWARRIOR_FOOT_GAPS.SHADOWWARRIOR_ATTACK_CONTENT_H[i],
		})
	return anchors
